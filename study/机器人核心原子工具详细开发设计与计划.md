# 机器人核心原子工具详细开发设计与计划

## 1. 概述

### 1.1 项目背景

OpenClaw是一个开源的多模态机器人控制平台，提供了丰富的技能生态系统。为了将机器人控制功能完全融入OpenClaw的核心工具集合，我们需要开发一系列符合PiClaw设计理念的机器人核心原子工具。

### 1.2 设计理念

完全基于 `/home/cc/work_dev/test/OpenClaw/1.md` 和 `/home/cc/work_dev/test/OpenClaw/2.md` 的设计理念：

- **不修改原有核心工具**：保持read、write、edit、bash工具的完整性和核心地位
- **新增机器人专用工具**：创建robot_read、robot_edit、robot_exec、robot_write四个机器人专用工具
- **PCR-URI统一资源标识**：使用`pcr://<resource_type>/<resource_subtype>/<resource_id>/<resource_attribute>`格式统一标识所有机器人资源
- **极简标准化设计**：符合Pi的"最小完备集"铁律
- **完全可扩展架构**：不修改OpenClaw核心框架，仅通过可扩展机制集成

## 3. OpenClaw核心工具系统架构

### 3.1 工具创建与集成流程

OpenClaw的工具系统遵循分层架构，核心工具通过`createOpenClawCodingTools()`函数统一创建和管理：

```typescript
// 核心工具创建函数 - src/agents/pi-tools.ts
export function createOpenClawCodingTools(options?: {
  agentId?: string;
  exec?: ExecToolDefaults & ProcessToolDefaults;
  messageProvider?: string;
  agentAccountId?: string;
  messageTo?: string;
  messageThreadId?: string | number;
  sandbox?: SandboxContext | null;
  sessionKey?: string;
  agentDir?: string;
  workspaceDir?: string;
  config?: OpenClawConfig;
  abortSignal?: AbortSignal;
  modelProvider?: string;
  modelId?: string;
  modelContextWindowTokens?: number;
  modelAuthMode?: ModelAuthMode;
  currentChannelId?: string;
  currentThreadTs?: string;
  currentMessageId?: string | number;
  groupId?: string | null;
  groupChannel?: string | null;
  groupSpace?: string | null;
  spawnedBy?: string | null;
  senderId?: string | null;
  senderName?: string | null;
  senderUsername?: string | null;
  senderE164?: string | null;
  replyToMode?: "off" | "first" | "all";
  hasRepliedRef?: { value: boolean };
  modelHasVision?: boolean;
  requireExplicitMessageTarget?: boolean;
  disableMessageTool?: boolean;
  senderIsOwner?: boolean;
}): AnyAgentTool[] {
  // 1. 配置解析与策略处理
  const { agentId, globalPolicy, agentPolicy, groupPolicy } = resolveEffectiveToolPolicy(options);

  // 2. 文件系统访问控制
  const fsPolicy = createToolFsPolicy({
    workspaceOnly: resolveToolFsConfig(options).workspaceOnly
  });

  // 3. 基础工具创建
  const base = (codingTools as unknown as AnyAgentTool[]).flatMap((tool) => {
    // 处理read工具
    if (tool.name === readTool.name) {
      if (sandboxRoot) {
        return [createSandboxedReadTool({ root: sandboxRoot, bridge: sandboxFsBridge! })];
      }
      return [createHostWorkspaceReadTool(workspaceRoot)];
    }

    // 处理write工具
    if (tool.name === "write") {
      if (sandboxRoot) {
        return [createSandboxedWriteTool({ root: sandboxRoot, bridge: sandboxFsBridge! })];
      }
      return [createHostWorkspaceWriteTool(workspaceRoot)];
    }

    // 处理edit工具
    if (tool.name === "edit") {
      if (sandboxRoot) {
        return [createSandboxedEditTool({ root: sandboxRoot, bridge: sandboxFsBridge! })];
      }
      return [createHostWorkspaceEditTool(workspaceRoot)];
    }

    // 忽略原有的bash/exec工具，我们将创建机器人专用版本
    if (tool.name === "bash" || tool.name === "exec") {
      return [];
    }

    return [tool];
  });

  // 4. 机器人专用工具创建
  const robotTools = [
    createRobotReadTool({
      config: options.config,
      agentId: agentId
    }),
    createRobotWriteTool({
      config: options.config,
      agentId: agentId
    }),
    createRobotEditTool({
      config: options.config,
      agentId: agentId
    }),
    createRobotExecTool({
      config: options.config,
      agentId: agentId
    })
  ];

  // 5. 工具策略应用
  const tools = [
    ...base,
    ...robotTools,
    ...createOpenClawTools(options)  // 集成其他OpenClaw工具
  ];

  // 6. 安全策略应用
  const authorized = applyOwnerOnlyToolPolicy(tools, options.senderIsOwner);
  return applyToolPolicyPipeline(authorized, policies);
}
```

### 3.2 PCR-URI解析器实现

```typescript
// src/agents/robot-tools/pcr-uri-parser.ts
export class PcrUriParser {
  static parse(uri: string): {
    resource_type: string;
    resource_subtype: string;
    resource_id: string;
    resource_attribute: string;
  } {
    const match = uri.match(/^pcr:\/\/([^/]+)\/([^/]+)\/([^/]+)\/(.+)$/);
    if (!match) {
      throw new ToolInputError(`Invalid PCR-URI format: ${uri}`);
    }
    return {
      resource_type: match[1],
      resource_subtype: match[2],
      resource_id: match[3],
      resource_attribute: match[4]
    };
  }
}
```

```typescript
async function writeRobotResource(uri: string, content: string): Promise<unknown> {
  const scheduler = ResourceScheduler.getInstance();

        // 如果解析失败，保留原始字符串
```

```typescript
async function executeRobotEdit(params: RobotEditToolParams): Promise<unknown> {
  const { uri, oldValue, newValue } = params;
  const scheduler = ResourceScheduler.getInstance();

  const currentValue = await scheduler.dispatch(uri, { action: "read" });
```

#### 3.4.4 RobotExecTool

```typescript
// src/agents/robot-tools/robot-exec.ts
import { createExecTool } from "../bash-tools.js";
import type { ExecToolDefaults } from "../bash-tools.exec.js";
import type { OpenClawConfig } from "../config/config.js";

export interface RobotExecToolParams {
  command: string;
  params?: Record<string, unknown>;
  timeout?: number;
}

export function createRobotExecTool(options?: {
  config?: OpenClawConfig;
  agentId?: string;
}): AnyAgentTool {
  const base = createExecTool({
    ...(options?.config?.tools?.exec),
    agentId: options?.agentId
  });

  return {
    ...base,
    name: "robot_exec",
    label: "Robot Exec",
    description: "Execute robot system commands or actions",
    parameters: {
      type: "object",
      properties: {
        command: {
          type: "string",
          description: "Robot command to execute"
        },
        params: {
          type: "object",
          description: "Command parameters"
        },
        timeout: {
          type: "number",
          description: "Execution timeout in milliseconds"
        }
      },
      required: ["command"]
    },
    execute: async (toolCallId: string, params: RobotExecToolParams, signal: AbortSignal) => {
      try {
        return await executeRobotCommand(params);
      } catch (error) {
        return errorResult(`Failed to execute robot command: ${error.message}`);
      }
    }
  };
}

async function executeRobotCommand(params: RobotExecToolParams): Promise<AgentToolResult> {
  const { command, params: cmdParams } = params;

  // 特殊处理机器人专用命令
  switch (command.toLowerCase()) {
    case "calibrate":
      return await calibrateRobot(cmdParams);
    case "reset":
      return await resetRobot(cmdParams);
    case "diagnostics":
      return await runDiagnostics(cmdParams);
    case "emergency_stop":
      return await emergencyStop(cmdParams);
    default:
      throw new ToolInputError(`Unknown robot command: ${command}`);
  }
}

async function calibrateRobot(params?: Record<string, unknown>): Promise<AgentToolResult> {
  const result = await runRobotAction("calibrate", params);
  return jsonResult({ status: "calibration_complete", data: result });
}

async function resetRobot(params?: Record<string, unknown>): Promise<AgentToolResult> {
  const result = await runRobotAction("reset", params);
  return jsonResult({ status: "reset_complete", data: result });
}

async function runDiagnostics(params?: Record<string, unknown>): Promise<AgentToolResult> {
  const result = await runRobotAction("diagnostics", params);
  return jsonResult({ status: "diagnostics_complete", data: result });
}

async function emergencyStop(params?: Record<string, unknown>): Promise<AgentToolResult> {
  const result = await runRobotAction("emergency_stop", params);
  return jsonResult({ status: "emergency_stopped", data: result });
}

async function runRobotAction(action: string, params?: Record<string, unknown>): Promise<unknown> {
  const uri = `pcr://system/exec/${action}`;
  const scheduler = ResourceScheduler.getInstance();
  return await scheduler.dispatch(uri, params || {});
}
```
